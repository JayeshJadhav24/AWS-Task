Project Overview
This hands-on lab guides you through building a fully functional TODO web application using FastAPI (Python) as the backend, with AWS services simulated locally using LocalStack. By the end you will have a running TODO app that creates and stores tasks, uploads attachments to S3, and sends email notifications via SES — all running on your local machine before deploying to real AWS.
What You Will Build
A FastAPI backend with full CRUD for TODO items
PostgreSQL database to store todos persistently
AWS S3 (via LocalStack) to store file attachments for todos
AWS SES (via LocalStack) to send email notifications when a todo is created
A simple HTML frontend to interact with the API
Everything running locally using LocalStack — no real AWS charges
What is LocalStack?
LocalStack is a tool that runs a fully functional AWS cloud environment on your local machine inside Docker. Instead of connecting to real AWS (which costs money and requires internet), your app connects to LocalStack which behaves exactly like AWS. This means you can develop and test S3 uploads, SES emails, and other AWS services completely for free on your laptop.
LocalStack simulates these AWS services locally:
S3      — File/object storage (todo attachments)
SES     — Simple Email Service (notifications)
EC2     — Not needed locally; you will deploy to real EC2 at the end
And 60+ other AWS services all running in one Docker container

# Each request gets its own session
SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)
# Base class for all database models
Base = declarative_base()
# Dependency — gives each API route a database session
# and automatically closes it when the request is done
def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

3.5 Create CRUD Operations
Create app/crud.py — this file contains all the database read/write logic:
# app/crud.py
from sqlalchemy.orm import Session
from app import models, schemas
def get_todos(db: Session, skip: int = 0, limit: int = 100):
    """Get all todos from the database"""
    return db.query(models.Todo).offset(skip).limit(limit).all()
def get_todo(db: Session, todo_id: int):
    """Get a single todo by ID"""
    return db.query(models.Todo).filter(models.Todo.id == todo_id).first()
def create_todo(db: Session, todo: schemas.TodoCreate):
    """Create a new todo in the database"""
    db_todo = models.Todo(**todo.model_dump())
    db.add(db_todo)
    db.commit()       # Save to database
    db.refresh(db_todo)  # Get the saved object with ID
    return db_todo
def update_todo(db: Session, todo_id: int, todo: schemas.TodoUpdate):
    """Update an existing todo"""
    db_todo = get_todo(db, todo_id)
    if not db_todo:
        return None
    # Only update fields that were provided
    update_data = todo.model_dump(exclude_unset=True)
    for field, value in update_data.items():
        setattr(db_todo, field, value)
    db.commit()
    db.refresh(db_todo)
    return db_todo
def delete_todo(db: Session, todo_id: int):
    """Delete a todo"""
    db_todo = get_todo(db, todo_id)
    if not db_todo:
        return None
    db.delete(db_todo)
    db.commit()
    return db_todo
def update_todo_attachment(db: Session, todo_id: int, attachment_url: str):
    """Save the S3 attachment URL to a todo"""
    db_todo = get_todo(db, todo_id)
    if db_todo:
        db_todo.attachment = attachment_url
        db.commit()
        db.refresh(db_todo)
    return db_todo

4.2 Create S3 Service
Create app/services/s3_service.py — this handles all file uploads to S3:
# app/services/s3_service.py
import boto3
import os
from dotenv import load_dotenv
from fastapi import UploadFile
import uuid
load_dotenv()
# Create S3 client pointing to LocalStack instead of real AWS
s3_client = boto3.client(
    's3',
    endpoint_url=os.getenv('AWS_ENDPOINT_URL'),  # http://localhost:4566
    aws_access_key_id=os.getenv('AWS_ACCESS_KEY_ID'),
    aws_secret_access_key=os.getenv('AWS_SECRET_ACCESS_KEY'),
    region_name=os.getenv('AWS_DEFAULT_REGION')
)
BUCKET_NAME = os.getenv('S3_BUCKET_NAME')
def create_bucket_if_not_exists():
    """Create S3 bucket if it does not already exist"""
    try:
        s3_client.head_bucket(Bucket=BUCKET_NAME)
        print(f'Bucket {BUCKET_NAME} already exists')
    except Exception:
        s3_client.create_bucket(Bucket=BUCKET_NAME)
        print(f'Created bucket: {BUCKET_NAME}')
def upload_file(file: UploadFile) -> str:
    """
    Upload a file to S3 and return its URL.
    The URL is what we store in the database.
    """
    # Generate a unique filename to avoid conflicts
    file_extension = file.filename.split('.')[-1]
    unique_filename = f'{uuid.uuid4()}.{file_extension}'
    # Upload the file to S3
    s3_client.upload_fileobj(
        file.file,
        BUCKET_NAME,
        unique_filename,
        ExtraArgs={'ContentType': file.content_type}
    )
    # Build the URL where the file can be accessed
    url = f'{os.getenv("AWS_ENDPOINT_URL")}/{BUCKET_NAME}/{unique_filename}'
    return url
def delete_file(file_url: str):
    """Delete a file from S3 using its URL"""
    # Extract the filename from the URL
    filename = file_url.split('/')[-1]
    s3_client.delete_object(Bucket=BUCKET_NAME, Key=filename)
4.3 Create SES Service
Create app/services/ses_service.py — this handles sending emails via SES:
# app/services/ses_service.py
import boto3
import os
from dotenv import load_dotenv
load_dotenv()
# Create SES client pointing to LocalStack
ses_client = boto3.client(
    'ses',
    endpoint_url=os.getenv('AWS_ENDPOINT_URL'),
    aws_access_key_id=os.getenv('AWS_ACCESS_KEY_ID'),
    aws_secret_access_key=os.getenv('AWS_SECRET_ACCESS_KEY'),
    region_name=os.getenv('AWS_DEFAULT_REGION')
)
SENDER = os.getenv('SES_SENDER_EMAIL')
def verify_email(email: str):
    """
    In real AWS SES you must verify emails before sending.
    LocalStack accepts any email but we still call this
    to mirror real AWS behavior.
    """
    ses_client.verify_email_identity(EmailAddress=email)
def send_todo_created_email(recipient_email: str, todo_title: str, todo_id: int):
    """
    Send an email notification when a new todo is created.
    In LocalStack this will not actually send a real email
    but the API call will succeed exactly like real AWS.
    """
    try:
        # First verify sender email (LocalStack requirement)
        verify_email(SENDER)
        verify_email(recipient_email)
        response = ses_client.send_email(
            Source=SENDER,
            Destination={
                'ToAddresses': [recipient_email]
            },
            Message={
                'Subject': {
                    'Data': f'New TODO Created: {todo_title}',
                    'Charset': 'UTF-8'
                },
                'Body': {
                    'Html': {
                        'Data': f'''
                            <h2>Your TODO has been created!</h2>
                            <p><b>Title:</b> {todo_title}</p>
                            <p><b>ID:</b> {todo_id}</p>
                            <p>Log in to your TODO app to manage this task.</p>
                        ''',
                        'Charset': 'UTF-8'
                    }
                }
            }
        )
        print(f'Email sent! Message ID: {response["MessageId"]}')
        return True
    except Exception as e:
        print(f'Email sending failed: {e}')
        return False
Chapter 5: FastAPI Routes
5.1 Create the Todo Router
Create app/routers/todos.py — this is where all the API endpoints are defined:
# app/routers/todos.py
from fastapi import APIRouter, Depends, HTTPException, UploadFile, File, Form
from sqlalchemy.orm import Session
from typing import List, Optional
from app import crud, schemas
from app.database import get_db
from app.services import s3_service, ses_service
router = APIRouter(prefix='/todos', tags=['todos'])
# -- GET all todos -----------------------------------
@router.get('/', response_model=List[schemas.TodoResponse])
def get_todos(skip: int = 0, limit: int = 100, db: Session = Depends(get_db)):
    """Get all todos. Supports pagination with skip and limit."""
    todos = crud.get_todos(db, skip=skip, limit=limit)
    return todos
# -- GET single todo ---------------------------------
@router.get('/{todo_id}', response_model=schemas.TodoResponse)
def get_todo(todo_id: int, db: Session = Depends(get_db)):
    """Get a single todo by its ID."""
    todo = crud.get_todo(db, todo_id)
    if not todo:
        raise HTTPException(status_code=404, detail='Todo not found')
    return todo
# -- CREATE todo -------------------------------------
@router.post('/', response_model=schemas.TodoResponse, status_code=201)
def create_todo(todo: schemas.TodoCreate, db: Session = Depends(get_db)):
    """
    Create a new todo.
    If an email is provided, sends a notification via SES.
    """
    db_todo = crud.create_todo(db, todo)
    # Send email notification if email was provided
    if todo.email:
        ses_service.send_todo_created_email(
            recipient_email=todo.email,
            todo_title=todo.title,
            todo_id=db_todo.id
        )
    return db_todo
# -- UPDATE todo -------------------------------------
@router.patch('/{todo_id}', response_model=schemas.TodoResponse)
def update_todo(todo_id: int, todo: schemas.TodoUpdate, db: Session = Depends(get_db)):
    """Update a todo. Only provided fields are updated."""
    db_todo = crud.update_todo(db, todo_id, todo)
    if not db_todo:
        raise HTTPException(status_code=404, detail='Todo not found')
    return db_todo
# -- DELETE todo -------------------------------------
@router.delete('/{todo_id}')
def delete_todo(todo_id: int, db: Session = Depends(get_db)):
    """Delete a todo by ID."""
    db_todo = crud.delete_todo(db, todo_id)
    if not db_todo:
        raise HTTPException(status_code=404, detail='Todo not found')
    return {'message': f'Todo {todo_id} deleted successfully'}
# -- UPLOAD attachment --------------------------------
@router.post('/{todo_id}/attachment', response_model=schemas.TodoResponse)
async def upload_attachment(
    todo_id: int,
    file: UploadFile = File(...),
    db: Session = Depends(get_db)
):
    """
    Upload a file attachment to a todo.
    The file is stored in S3 and the URL is saved in the database.
    """
    # Check todo exists
    db_todo = crud.get_todo(db, todo_id)
    if not db_todo:
        raise HTTPException(status_code=404, detail='Todo not found')
    # Upload file to S3
    file_url = s3_service.upload_file(file)
    # Save the S3 URL in the database
    db_todo = crud.update_todo_attachment(db, todo_id, file_url)
    return db_todo

      <option value='low'>Low Priority</option>
      <option value='medium' selected>Medium Priority</option>
      <option value='high'>High Priority</option>
    </select>
    <input id='email' placeholder='Email for notification (optional)' type='email' />
    <button onclick='createTodo()'>Create TODO</button>
  </div>
  <h2>All TODOs</h2>
  <button onclick='loadTodos()' style='margin-bottom:10px'>Refresh List</button>
  <div id='todos-list'></div>
  <script>
    const API = 'http://localhost:8000';
    async function loadTodos() {
      const res  = await fetch(`${API}/todos/`);
      const todos = await res.json();
      const div  = document.getElementById('todos-list');
      if (todos.length === 0) {
        div.innerHTML = '<p>No todos yet. Create one above!</p>';
        return;
      }
      div.innerHTML = todos.map(t => `
        <div class='todo-item ${t.completed ? 'completed' : ''} priority-${t.priority}'>
          <strong>${t.id}. ${t.title}</strong>
          <span style='float:right;font-size:12px'>${t.priority.toUpperCase()}</span>
          <p>${t.description || 'No description'}</p>
          ${t.attachment ? `<a href='${t.attachment}' target='_blank'>View Attachment</a>` : ''}
          <br>
          <button onclick='toggleComplete(${t.id}, ${t.completed})'>
            ${t.completed ? 'Mark Incomplete' : 'Mark Complete'}
          </button>
          <button onclick='deleteTodo(${t.id})'
            style='background:#DC2626;margin-left:8px'>Delete</button>
          <label style='margin-left:8px'>
            Upload file:
            <input type='file' onchange='uploadFile(${t.id}, this)' />
          </label>
        </div>
      `).join('');
    }
    async function createTodo() {
      const title = document.getElementById('title').value;
      if (!title) { alert('Title is required!'); return; }
      await fetch(`${API}/todos/`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          title,
          description: document.getElementById('description').value,
          priority:    document.getElementById('priority').value,
          email:       document.getElementById('email').value || null
        })
      });
      document.getElementById('title').value = '';
      document.getElementById('description').value = '';
      document.getElementById('email').value = '';
      loadTodos();
    }
    async function toggleComplete(id, currentStatus) {
      await fetch(`${API}/todos/${id}`, {
        method: 'PATCH',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ completed: !currentStatus })
      });
      loadTodos();
    }
    async function deleteTodo(id) {
      if (!confirm('Delete this todo?')) return;
      await fetch(`${API}/todos/${id}`, { method: 'DELETE' });
      loadTodos();
    }
    async function uploadFile(todoId, input) {
      const file = input.files[0];
      if (!file) return;
      const form = new FormData();
      form.append('file', file);
      await fetch(`${API}/todos/${todoId}/attachment`, {
        method: 'POST', body: form
      });
      loadTodos();
    }
    // Load todos when page opens
    loadTodos();
  </script>
</body>

9.1 Launch EC2 Instance
Go to AWS Console ? EC2 ? Launch Instance
Name: todo-app-server

Instance type: t3.micro (Free tier eligible)
Create a new key pair ? name it todo-key ? download the .pem file

Click Launch Instance
9.2 Connect to EC2
# Change key file permissions (Mac/Linux only)
chmod 400 todo-key.pem
# Connect via SSH (replace with your EC2 public IP)
ssh -i todo-key.pem ubuntu@YOUR_EC2_PUBLIC_IP
9.3 Install Dependencies on EC2
# Update system packages
sudo apt update && sudo apt upgrade -y
# Install Python and pip
sudo apt install python3 python3-pip python3-venv git -y
# Install PostgreSQL client
sudo apt install postgresql-client -y
9.4 Deploy the Application
# Clone your project (or use scp to copy files)
git clone https://github.com/YOUR_USERNAME/todo-app.git
cd todo-app
# Create virtual environment
python3 -m venv venv
source venv/bin/activate
# Install dependencies
pip install -r requirements.txt
# Create .env file with REAL AWS credentials this time
# (Remove the AWS_ENDPOINT_URL line — use real AWS)
nano .env
9.5 Update .env for Real AWS
# .env on EC2 — uses REAL AWS services (no LocalStack)
DATABASE_URL=postgresql://postgres:password@YOUR_RDS_ENDPOINT:5432/tododb

AWS_ACCESS_KEY_ID=YOUR_REAL_ACCESS_KEY
